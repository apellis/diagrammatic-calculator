# The golden corpus: terms, their normal forms, and (in actions.json) their polynomial action.
#
# The Lean program lean/golden/Golden.lean reads this file, computes the action of each term
# and of its normal form on every monomial x^a (each a_k < n) at every source idempotent, by
# composing the generators' defining operators, checks that the two agree, and writes
# actions.json. The engine must reproduce both the normal form text and the action exactly.
#
# One case per block: case ID, context, term, nf (normal form). Terms are written expanded.

case nh2-dot-slide
context nilhecke nu=2
term x_1 psi_1
nf 1 + psi_1 x_2

case nh2-crossing-dots
context nilhecke nu=2
term psi_1 x_1^3
nf psi_1 x_1^3

case nh3-braid
context nilhecke nu=3
term psi_1 psi_2 psi_1 - psi_2 psi_1 psi_2
nf 0

case nh3-mixed
context nilhecke nu=3
term x_3 psi_2 psi_1 x_1
nf -psi_1 x_1 - x_1 psi_2 + psi_2 psi_1 x_1^2

case nh3-idempotent
context nilhecke nu=3
term psi_1 psi_2 psi_1 x_1^2 x_2 psi_1 psi_2 psi_1
nf psi_1 psi_2 psi_1

case onh2-dot
context odd-nilhecke nu=2
term x_1 psi_1
nf x_1 psi_1

case onh2-dots-through
context odd-nilhecke nu=2
term psi_1 x_2 x_1
nf x_1 x_2 psi_1

case onh3-longest
context odd-nilhecke nu=3
term psi_1 psi_2 psi_1 x_1^2 x_2
nf -1 + x_2 psi_1 + psi_1 x_3 + x_3 psi_2 + x_3^2 psi_2 psi_1 - x_2 psi_1 x_3 psi_2 + x_2 x_3 psi_1 x_3 psi_2 psi_1

case onh3-mixed
context odd-nilhecke nu=3
term psi_2 x_1 psi_1 x_3 psi_2
nf x_1 psi_1 psi_2 - x_1 x_2 psi_1 psi_2 psi_1

case onh4-anticommute
context odd-nilhecke nu=4
term psi_1 psi_3 + psi_3 psi_1
nf 0

case onh4-distant
context odd-nilhecke nu=4
term psi_3 x_1 psi_1 x_4
nf x_1 psi_1 + x_1 x_3 psi_1 psi_3

case a2-square
context klr type=A2 nu=i+j
term psi_1 psi_1 e(ij)
nf x_1 e(ij) + x_2 e(ij)

case a2-braid
context klr type=A2 nu=2i+j
term psi_1 psi_2 psi_1 e(iji) - psi_2 psi_1 psi_2 e(iji)
nf e(iji)

case a2-iij
context klr type=A2 nu=2i+j
term psi_2 psi_1 x_1 psi_2 e(iij)
nf psi_1 psi_2 psi_1 x_1 e(iij)

case a2-jii
context klr type=A2 nu=2i+j
term x_3 psi_1 psi_2 psi_1 e(jii)
nf psi_1 psi_2 psi_1 x_1 e(jii)

case b2-square-ij
context klr type=B2 nu=i+j
term psi_1 psi_1 e(ij)
nf x_1 e(ij) + x_2^2 e(ij)

case b2-square-ji
context klr type=B2 nu=i+j
term psi_1 psi_1 e(ji)
nf x_2 e(ji) + x_1^2 e(ji)

case b2-braid-iji
context klr type=B2 nu=2i+j
term psi_1 psi_2 psi_1 e(iji) - psi_2 psi_1 psi_2 e(iji)
nf e(iji)

case b2-braid-jij
context klr type=B2 nu=i+2j
term psi_1 psi_2 psi_1 e(jij) - psi_2 psi_1 psi_2 e(jij)
nf x_1 e(jij) + x_3 e(jij)

case b2-iij
context klr type=B2 nu=2i+j
term x_2 psi_1 psi_2 x_3 e(iij)
nf psi_1 psi_2 x_1 x_3 e(iij)

# Weighted KLR algebras (contexts `klrw`, docs/klrw.md): loadings are compared up to isotopy.

case klrw-a1-dot-crossing
context klrw type=A1
term psi_1 x_1 e(i@0 i@1)
nf b(2 1; i@0 i@1) x_1 e(i@0 i@1)

case klrw-a1-same-colour-bigon
context klrw type=A1
term psi_1 psi_1 e(i@0 i@1)
nf 0

case klrw-a1-basis-dot-slide
context klrw type=A1
term b(2 3 1; i@0 i@1 i@2) x_3 e(i@0 i@1 i@2) - x_1 b(2 3 1; i@0 i@1 i@2) e(i@0 i@1 i@2)
nf -b(1 3 2; i@0 i@1 i@2) e(i@0 i@1 i@2) - b(2 1 3; i@0 i@1 i@2) e(i@0 i@1 i@2)

case klrw-a2-braid
context klrw type=A2
term psi_2 psi_1 psi_2 e(i@0 j@1 i@2) - psi_1 psi_2 psi_1 e(i@0 j@1 i@2)
nf -e(i@0 j@1 i@2)

case klrw-a2-weighted-bigon-dot
context klrw type=A2 theta_ij=1/2
term psi_1 x_2 psi_1 e(i@0 j@1)
nf x_1^2 e(i@0 j@1) + x_1 x_2 e(i@0 j@1)

case klrw-a2-ghost-bigon-dots
context klrw type=A2 theta_ij=1/2
term x_1 to(j@0 i@1) x_2 to(j@0 i@1/4) e(j@0 i@1)
nf x_1^2 x_2 e(j@0 i@1) + x_1 x_2^2 e(j@0 i@1)

case klrw-a2-ghost-bigon-equals-crossing-bigon
context klrw type=A2 theta_ij=-1/2
term to(i@0 j@1) to(i@0 j@1/4) e(i@0 j@1) - psi_1 psi_1 e(i@0 j@1)
nf 0

case klrw-a2-weighted-braid
context klrw type=A2 theta_ij=1/2
term psi_2 psi_1 psi_2 e(j@0 i@1 j@2) - psi_1 psi_2 psi_1 e(j@0 i@1 j@2)
nf -e(j@0 i@1 j@2)

case klrw-b2-ghost-bigon
context klrw type=B2 theta_ij=1/2
term to(j@0 i@1) to(j@0 i@1/4) e(j@0 i@1)
nf x_1^2 e(j@0 i@1) + x_2 e(j@0 i@1)

case klrw-b2-weighted-bigon
context klrw type=B2 theta_ij=1/3
term psi_1 x_1 psi_1 e(j@0 i@1) + 2 x_2 e(j@0 i@1)
nf x_1^2 x_2 e(j@0 i@1) + x_2^2 e(j@0 i@1) + 2 x_2 e(j@0 i@1)

case klrw-a1-red-bigon
context klrw type=A1 red="2i@0"
term to(i@-1) to(i@1) e(i@-1)
nf x_1^2 e(i@-1)

case klrw-a1-red-bigon-two-strands
context klrw type=A1 red="i@0"
term to(i@1/2 i@1) to(i@-1 i@-1/2) e(i@1/2 i@1)
nf x_1 x_2 e(i@1/2 i@1)

case klrw-a1-red-crossings
context klrw type=A1 red="i@0"
term psi_1 to(i@-1/2 i@1) psi_1 e(i@-1/2 i@1)
nf b(2 1; i@-1/2 i@1) e(i@-1/2 i@1)

case klrw-a2-two-red-strands
context klrw type=A2 red="i@0, j@2"
term to(i@1 j@3) to(i@-1 j@1) e(i@1 j@3)
nf x_1 x_2 e(i@1 j@3)

case klrw-a2-red-weighted
context klrw type=A2 red="i+j@0" theta_ij=1/2
term to(j@1 i@2) x_2 to(j@-1 i@1) e(j@1 i@2)
nf x_1 x_2 e(j@1 i@2)

case klrw-a2-red-basis
context klrw type=A2 red="i@0" theta_ij=1/2
term b(2 1; j@-1 i@1) x_1 e(i@-1 j@1/4)
nf b(2 1; j@-1 i@1) x_1 e(i@-1 j@1/4)
